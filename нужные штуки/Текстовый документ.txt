4. reset-light: <!doctype html>
<html lang="ru" data-theme="light">
<head>
  <meta charset="utf-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1" />
  <title>Derma Mini-App — Обнуление данных (Light)</title>

<script src="js/app-state.js"></script>
<script>
  // Инициализация всех переключателей темы на странице
  document.addEventListener('DOMContentLoaded', () => {
    const settings = AppState.get('settings') || {};
    const theme = (settings.theme === 'light') ? 'light' : 'dark';
    document.documentElement.dataset.theme = theme;

    // найдём все тумблеры с data-role="theme-switch"
    document.querySelectorAll('[data-role="theme-switch"]').forEach(sw => {
      const isLight = (document.documentElement.dataset.theme === 'light');
      sw.setAttribute('aria-pressed', isLight ? 'true' : 'false');

      sw.addEventListener('click', () => {
        const cur = document.documentElement.dataset.theme === 'light' ? 'light' : 'dark';
        const next = (cur === 'light') ? 'dark' : 'light';
        AppState.applyTheme(next);                  // сохраняем и ставим data-theme
        // визуально обновим все тумблеры на странице
        document.querySelectorAll('[data-role="theme-switch"]')
          .forEach(x => x.setAttribute('aria-pressed', next === 'light' ? 'true' : 'false'));
      });
    });
  });
</script>
